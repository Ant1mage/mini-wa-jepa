import os,time,ctypes as C,ctypes.util
os.environ['SDL_VIDEODRIVER']='x11'
import pygame
from scripts.probe_keyboard_desktop import DesktopKeys
pygame.display.init();pygame.display.set_mode((600,220));pygame.display.set_caption('mini-wa-jepa | isolated controls')
time.sleep(.5)
print('driver',pygame.display.get_driver(),'wm',pygame.display.get_wm_info(),flush=True)
print('initial',pygame.event.get(),flush=True)
k=DesktopKeys();print('target',k.window,flush=True)
k.tap('space');time.sleep(.2);print('send_events',pygame.event.get(),flush=True)
x=C.CDLL(ctypes.util.find_library('Xtst'));x.XTestFakeKeyEvent.argtypes=[C.c_void_p,C.c_uint,C.c_int,C.c_ulong]
k.x.XSetInputFocus(k.display,k.window,2,0);k.x.XSync(k.display,False)
code=k.x.XKeysymToKeycode(k.display,k.x.XStringToKeysym(b'space'));print('code',code,flush=True)
x.XTestFakeKeyEvent(k.display,code,1,0);k.x.XSync(k.display,False);time.sleep(.2)
print('xtest_down',pygame.event.get(),flush=True)
x.XTestFakeKeyEvent(k.display,code,0,0);k.x.XSync(k.display,False);time.sleep(.2)
print('xtest_up',pygame.event.get(),flush=True)
k.close();pygame.quit()
